\section{Why CSDR?}\label{why-csdr}

\begin{frame}{What do we want to learn from a mixture?}
\phantomsection\label{what-do-we-want-to-learn-from-a-mixture}
\begin{columns}[T]
\begin{column}{0.52\linewidth}
\begin{block}{Scientific goals}
\phantomsection\label{scientific-goals}
\begin{itemize}
\tightlist
\item
  Estimate the \textbf{joint health effect}
\item
  Learn the \textbf{shape} of the response
\item
  Identify \textbf{important contributors}
\item
  Find a useful \textbf{mixture summary}
\end{itemize}

\vspace{0.03cm}
\end{block}
\end{column}

\begin{column}{0.44\linewidth}
\begin{alertblock}{The recurring problem}
\phantomsection\label{the-recurring-problem}
Many correlated exposures and confounding---all at once.

\[
(X_1,\ldots,X_p)\longrightarrow Y
\]
\end{alertblock}
\end{column}
\end{columns}

\muted{Different methods prioritize different scientific questions.}
\end{frame}

\begin{frame}{Mixtures methods make different bargains}
\phantomsection\label{mixtures-methods-make-different-bargains}

\begin{alertblock}{Key distinction}
\phantomsection\label{key-distinction}
CSDR learns the structure in \(X\) needed to preserve the
\textbf{causal} mean response.
\end{alertblock}
\end{frame}

\begin{frame}{Why not always estimate the full surface?}
\phantomsection\label{why-not-always-estimate-the-full-surface}
\begin{columns}[T]
\begin{column}{0.3\linewidth}
\begin{block}{One exposure}
\phantomsection\label{one-exposure}
Many observations support a curve.

\[
\mu(x_1)
\]
\end{block}
\end{column}

\begin{column}{0.3\linewidth}
\begin{block}{Two exposures}
\phantomsection\label{two-exposures}
Observations thin across a surface.

\[
\mu(x_1,x_2)
\]
\end{block}
\end{column}

\begin{column}{0.34\linewidth}
\begin{block}{Many exposures}
\phantomsection\label{many-exposures}
Most joint exposure combinations are sparse.

\[
\mu(x_1,\ldots,x_p)
\]
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Consequence}
\phantomsection\label{consequence}
More dimensions mean harder nuisance estimation, weaker support, and a
response surface that is difficult to communicate.
\end{alertblock}
\end{frame}

\begin{frame}{CSDR searches for the intrinsic causal dimension}
\phantomsection\label{csdr-searches-for-the-intrinsic-causal-dimension}
\[
\mu(x)=\mathbb{E}(Y^x)=g(\beta^\top x),
\qquad
\beta\in\mathbb{R}^{p\times d},
\qquad
d\ll p.
\]

\vspace{0.20cm}

\begin{columns}[T]
\begin{column}{0.29\linewidth}
\begin{block}{Exposures}
\phantomsection\label{exposures}
\[
X_1,\ X_2,\ \ldots,\ X_p
\]
\end{block}
\end{column}

\begin{column}{0.33\linewidth}
\begin{block}{Causal summary}
\phantomsection\label{causal-summary}
\[
Z=\beta^\top X\in\mathbb{R}^d
\]
\end{block}
\end{column}

\begin{column}{0.32\linewidth}
\begin{block}{Flexible response}
\phantomsection\label{flexible-response}
\[
\mu_\beta(z)=g(z)
\]
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{Plain-language target}
\phantomsection\label{plain-language-target}
Replace \(p\) exposures with \(d\) summaries \textbf{without losing the
causal mean response}.
\end{alertblock}
\end{frame}

\begin{frame}{Causal directions can differ from associational
directions}
\phantomsection\label{causal-directions-can-differ-from-associational-directions}
\begin{columns}[T]
\begin{column}{0.47\linewidth}
\begin{block}{Observed regression}
\phantomsection\label{observed-regression}
\[
\mathbb{E}(Y\mid X=x)
\]

Confounding can create or obscure exposure directions associated with
the outcome.

\[
X\longleftarrow C\longrightarrow Y
\]
\end{block}
\end{column}

\begin{column}{0.47\linewidth}
\begin{block}{Causal response}
\phantomsection\label{causal-response}
\[
\mathbb{E}(Y^x)
\]

CSDR seeks directions that preserve the response under interventions on
\(X\).

\[
X\longrightarrow Y
\]
\end{block}
\end{column}
\end{columns}

\begin{alertblock}{The distinction that matters}
\phantomsection\label{the-distinction-that-matters}
\centering

\Large Association-preserving \(\boldsymbol{\neq}\)
causal-effect-preserving
\end{alertblock}
\end{frame}

\begin{frame}{A modular two-stage strategy}
\phantomsection\label{a-modular-two-stage-strategy}
\begin{columns}[T]
\begin{column}{0.31\linewidth}
\begin{block}{1. Adjust}
\phantomsection\label{adjust}
Use \((Y,X,C)\) to construct a cross-fitted causal response
\(\widetilde Y\).

\[
\mathbb{E}(\widetilde Y\mid X)=\mu(X)
\]
\end{block}
\end{column}

\begin{column}{0.31\linewidth}
\begin{block}{2. Reduce}
\phantomsection\label{reduce}
Apply sufficient dimension reduction to \((\widetilde Y,X)\).

\[
(\widehat\beta,\widehat d)
\]
\end{block}
\end{column}

\begin{column}{0.31\linewidth}
\begin{block}{3. Estimate}
\phantomsection\label{estimate}
Project exposures and estimate the causal response in \(d\) dimensions.

\[
\widehat Z=\widehat\beta^\top X
\]
\end{block}
\end{column}
\end{columns}

\vspace{0.25cm}

\centering

\method{Causal nuisance estimation} \(\longrightarrow\)
\method{standard SDR} \(\longrightarrow\) \method{reduced causal ERS}
\end{frame}

\begin{frame}{What does the researcher get?}
\phantomsection\label{what-does-the-researcher-get}
\begin{columns}[T]
\begin{column}{0.3\linewidth}
\begin{block}{Dimension}
\phantomsection\label{dimension}
\centering

\Huge \(\widehat d\)

\normalsize

How many causal directions are needed?
\end{block}
\end{column}

\begin{column}{0.32\linewidth}

\begin{frame}{Identification assumptions}
\phantomsection\label{identification-assumptions}
\begin{enumerate}
\tightlist
\item
  \textbf{Consistency:} observed outcomes agree with potential outcomes
  under the observed exposure.
\item
  \textbf{Conditional exchangeability:} \(Y^x\perp\!\!\!\perp X\mid C\).
\item
  \textbf{Positivity:} exposure values of interest are supported within
  levels of \(C\).
\end{enumerate}

\begin{alertblock}{Practical reminder}
\phantomsection\label{practical-reminder}
Dimension reduction may improve the support problem, but it does not
justify extrapolation or repair an inadequate adjustment set.
\end{alertblock}
\end{frame}
