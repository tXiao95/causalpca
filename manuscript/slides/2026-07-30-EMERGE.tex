\section{Why causal dimension

\begin{frame}{What do we want to learn from an environmental mixture?}
\phantomsection\label{what-do-we-want-to-learn-from-an-environmental-mixture}
\begin{columns}[T]
\begin{column}{0.45\linewidth}
\begin{block}{Scientific goals}
\phantomsection\label{scientific-goals}
\begin{itemize}
\tightlist
\item
  Estimate the \textbf{joint health effect}
\item
  Learn the \textbf{exposure-response curve}
\item
  Identify \textbf{important contributors}
\item
  Find a useful \textbf{mixture summary}
\end{itemize}

\vspace{0.03cm}
\end{block}
\end{column}

\begin{column}{0.45\linewidth}
\pause

\begin{alertblock}{The recurring problem}
\phantomsection\label{the-recurring-problem}
Highly correlated exposures, sparse support, and confounding---all at
once.

\[
(X_1,\ldots,X_p)\longrightarrow Y
\]
\end{alertblock}
\end{column}
\end{columns}

\vspace{0.4cm}

\pause

\begin{center}
\muted{Different methods prioritize different scientific goals.}
\end{center}
\end{frame}

\begin{frame}{Mixtures methods make different tradeoffs}
\phantomsection\label{mixtures-methods-make-different-tradeoffs}
\begin{center}
\scriptsize
\renewcommand{\arraystretch}{1.35}
\setlength{\tabcolsep}{4.5pt}
\begin{tabular}{@{}l c l l@{}}
\toprule
\textbf{Method} & \textbf{Dimension} & \textbf{Description} & \textbf{Natural strength} \\
\midrule
WQS / qgcomp & $d=1$ & Scalar joint quantile shift & Interpretable effect \\
PCA / PCP & $1\le d\le p$ & Unsupervised dimension reduction & Pattern discovery \\
BKMR / BART & $d=p$ & Full exposure-response & Nonlinear interactions \\
\textbf{CSDR} & \textbf{learn $1\le d\le p$} &
\textbf{Supervised causal dimension reduction} & \textbf{Flexible causal summary} \\
\bottomrule
\end{tabular}
\end{center}

\phantomsection\label{key-distinction}
CSDR estimates a \textbf{supervised causal dimension reduction} of the
exposure mixture \(X\) that preserves the \textbf{exposure-response
curve}.
\end{alertblock}
\end{frame}

\section{Conclusion}\label{conclusion}

\begin{frame}[fragile]{When should I use CSDR?}
\phantomsection\label{when-should-i-use-csdr}
\begin{columns}[T]
\begin{column}{0.6\linewidth}
\begin{block}{Use CSDR when you want}
\phantomsection\label{use-csdr-when-you-want}
\begin{itemize}
\tightlist
\item
  A joint causal exposure-response curve
\item
  Supervised dimension reduction with estimated \(d\)
\item
  Flexible modeling of complex relationships
\item
  Variable importance measures
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.36\linewidth}
\begin{block}{CSDR does not replace}
\phantomsection\label{csdr-does-not-replace}
\begin{itemize}
\tightlist
\item
  Good study design
\item
  Confounder selection
\item
  Adequate exposure support
\item
  Sensitivity analysis
\end{itemize}
\end{block}
\end{column}
\end{columns}

\pause

\begin{alertblock}{One-sentence takeaway}
\phantomsection\label{one-sentence-takeaway}
CSDR learns the smallest exposure representation \(Z=\beta^\top X\) that
preserves the exposure-response curve, and \texttt{csdr} turns that
target into an applied workflow.
\end{alertblock}

\begin{block}{Assumptions A1--A5 from the CSDR framework}
\begin{minipage}[t][4.15cm][t]{\linewidth}
\fontsize{6.6}{7.8}\selectfont
\setlength{\parskip}{0pt}
\setlength{\abovedisplayskip}{2.5pt}
\setlength{\belowdisplayskip}{2.5pt}
\setlength{\abovedisplayshortskip}{2pt}
\setlength{\belowdisplayshortskip}{2pt}

\begin{columns}[T,onlytextwidth]
\begin{column}{0.485\linewidth}
\raggedright
\textbf{\textcolor{emoryblue}{A1. Identification for \(X\)}}\par
\[
\begin{array}{@{}r@{\;}l@{\quad}l@{}}
\text{\bfseries a)} & \text{\bfseries Consistency:}
  & X=x\Longrightarrow Y=Y^x,\quad \forall x,\\
\text{\bfseries b)} & \text{\bfseries Strong positivity:}
  & \inf_{x,c}\pi(x\mid c)\ge\varepsilon>0,\\
\text{\bfseries c)} & \text{\bfseries Conditional ignorability:}
  & Y^x\perp\!\!\!\perp X\mid C,\quad \forall x .
\end{array}
\]

\vspace{0.10cm}
\textbf{\textcolor{emoryblue}{A2. Smoothness for continuous exposures}}\par
\(\,m(\cdot,c),\pi(\cdot\mid c)\in C^3(\mathcal X)\), with uniformly
bounded derivatives; the conditional variance and its \(x\)-derivatives are
uniformly bounded. The kernel \(k\) is bounded, differentiable, symmetric, and
second order:
\[
\int k(u)\,du=1,\qquad
\int u\,k(u)\,du=0,\qquad
\int u^2k(u)\,du<\infty .
\]
\end{column}

\begin{column}{0.485\linewidth}
\raggedright
\textbf{\textcolor{emoryblue}{A3. Existence of the CCMS}}\par
Let
\[
\mathcal A=
\left\{S:
\begin{array}{c}
\mu(x)=\mu(P_Sx)\\[-1pt]
\text{for all }x\in\mathcal X
\end{array}
\right\}.
\]
Then
\[
\mathcal S_{E(Y^x)}
=\bigcap_{S\in\mathcal A}S\in\mathcal A .
\]

\vspace{0.09cm}
\textbf{\textcolor{emoryblue}{A4. Well-defined reduced exposure}}\par
\vspace{-0.07cm}
\[
\beta^\top x=\beta^\top x'
\quad\Longrightarrow\quad
Y^x=Y^{x'}\ \text{a.s.}
\]

\vspace{0.01cm}
\textbf{\textcolor{emoryblue}{A5. Strong positivity for \(Z=\beta^\top X\)}}\par
\vspace{-0.08cm}
\[
\inf_{z,c}
\pi_\beta(z\mid c)\ge\varepsilon>0 .
\]
\end{column}
\end{columns}
\end{minipage}
\end{block}

\begin{frame}{Additional estimation assumptions}
\phantomsection\label{additional-estimation-assumptions}
\vspace{-0.43cm}

\begin{block}{Assumptions A6--A8 from the CSDR framework}
\begin{minipage}[t][5.48cm][t]{\linewidth}
\fontsize{6.5}{7.7}\selectfont
\setlength{\parskip}{0pt}
\setlength{\abovedisplayskip}{2.5pt}
\setlength{\belowdisplayskip}{2.5pt}
\setlength{\abovedisplayshortskip}{2pt}
\setlength{\belowdisplayshortskip}{2pt}

\vspace{0.08cm}
\begin{columns}[T,onlytextwidth]
\begin{column}{0.405\linewidth}
\raggedright
\textbf{\textcolor{emoryblue}{A6. Additive confounding model}}\par
There exist measurable functions \(g,h,f\) such that
\[
\begin{gathered}
Y=g(\beta^\top X)+h(C)+\varepsilon_Y,\\[-1pt]
\mathbb E(\varepsilon_Y\mid X,C)=0,\\[1pt]
X=f(C)+\varepsilon_X,\\[-1pt]
\mathbb E(\varepsilon_X\mid C)=0,\quad
\varepsilon_X\perp\!\!\!\perp C .
\end{gathered}
\]

\vspace{0.10cm}
\textbf{\textcolor{emoryblue}{A8. Uniform nonsingularity}}\par
For
\[
\mathcal X_{h,i}(\beta,x)=
\begin{pmatrix}
1\\ h^{-1}\beta^\top(X_i-x)
\end{pmatrix},
\]
define the local design matrix
\[
S_n(\beta,x)=\frac1n\sum_{i=1}^n
K_{h,i}(x)\,
\mathcal X_{h,i}(\beta,x)
\mathcal X_{h,i}(\beta,x)^\top .
\]
Assume
\[
\sup_{\beta,x}\left\|S_n(\beta,x)^{-1}\right\|=O_p(1).
\]
\end{column}

\begin{column}{0.555\linewidth}
\raggedright
\textbf{\textcolor{emoryblue}{A7. csMAVE regularity conditions}}\par
\textbf{(a)} \(X\) has compact support; its marginal density \(f_X\)
has bounded fourth derivative and is bounded away from zero.

\vspace{0.07cm}
\textbf{(b)} \(\mu(x)=g(\beta^\top x)\), where \(g\) has bounded,
continuous third derivatives.

\vspace{0.07cm}
\textbf{(c)} \(K\) is a spherically symmetric density with bounded
derivative and bounded support. All moments exist, and
\[
\int uu^\top K(u)\,du=I .
\]

\vspace{0.05cm}
\textbf{(d)} With
\(
V(z)=\partial g(z)/\partial z
\),
\[
\mathbb E\!\left[
V(\beta_0^\top X)V(\beta_0^\top X)^\top
\right]
\quad\text{is nonsingular}.
\]

\vspace{0.05cm}
\textbf{(e)} The first-stage ERS estimator satisfies
\[
\max_{1\le i\le n}
\left|\widehat\mu(X_i)-\mu(X_i)\right|
=O_p(R_n),
\qquad R_n=o(h^2).
\]
\end{column}
\end{columns}
\end{minipage}
\end{block}
\end{frame}

\begin{frame}{Theorem 1: recovering the causal subspace}
\phantomsection\label{theorem-1-recovering-the-causal-subspace}
\vspace{-0.16cm}

\begin{block}{Theorem 1 (Convergence of csMAVE)}
\begin{minipage}[t][5.10cm][t]{\linewidth}
\small
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
Let \(\beta_0\) be a basis for the true CCMS
\(\mathcal S_{\mathbb E(Y^x)}\), and let \(\widehat\beta\) minimize the
csMAVE objective using an estimated ERS response \(\widehat\mu(X)\). Define
\[
\delta_n=
\left\{\frac{\log n}{n h^p}\right\}^{1/2},
\qquad
\max_{1\le i\le n}
\left|\widehat\mu(X_i)-\mu(X_i)\right|
=O_p(R_n).
\]
Under Assumptions A1--A3, A7, and A8, if
\[
h\to0,\qquad d\ge d_0,\qquad
\frac{nh^p}{\log n}\to\infty,
\]
then
\[
\boxed{\;
\left\|
(I_p-\widehat\beta\widehat\beta^\top)\beta_0
\right\|
=
O_p\!\left(h^3+h\delta_n+h^{-1}R_n\right).
\;}
\]
\end{minipage}
\end{block}
\end{frame}

\begin{frame}{Theorem 2: learning the structural dimension}
\phantomsection\label{theorem-2-learning-the-structural-dimension}
\vspace{-0.16cm}

\begin{block}{Theorem 2 (Consistency of \(\widehat d\))}
\begin{minipage}[t][3.85cm][t]{\linewidth}
\small
For each candidate dimension \(d\in\{0,1,\ldots,p\}\), let
\(\widehat\beta_d\) be the corresponding csMAVE estimate and let
\(\operatorname{CV}(d)\) be the cross-validated prediction error for
\(\widehat\mu(X_i)\) using the reduced exposure
\(\widehat\beta_d^\top X_i\). Select
\[
\widehat d
=
\underset{d\in\{0,1,\ldots,p\}}{\arg\min}\;
\operatorname{CV}(d).
\]
Under Assumptions A1--A3 and A7,
\[
\boxed{\qquad
\Pr(\widehat d=d_0)\longrightarrow1
\qquad\text{as }n\to\infty .
\qquad}
\]
\end{minipage}
\end{block}
\end{frame}
