\section{Introduction}\label{introduction}

\begin{frame}{Existing approaches force a difficult tradeoff}
\phantomsection\label{existing-approaches-force-a-difficult-tradeoff}
\begin{itemize}
\tightlist
\item
  Single-index methods emphasize interpretability.
\item
  Full exposure-response methods emphasize flexibility.
\item
  \method{Causal SDR} targets a low-dimensional causal representation.
\end{itemize}
\end{frame}

\begin{frame}{The target is a causal subspace}
\phantomsection\label{the-target-is-a-causal-subspace}
\begin{block}{Definition}
\phantomsection\label{definition}
We seek a matrix \(\beta\in\mathbb{R}^{p\times d}\) such that

\[
\mu(x)=\mu_\beta(\beta^\top x), \qquad d\ll p.
\]
\end{block}

\begin{alertblock}{Key implication}
\phantomsection\label{key-implication}
The intervention-response surface depends on \(x\) only through
\focus{$\beta^\top x$}.
\end{alertblock}
\end{frame}

\begin{frame}{The first-stage error appears explicitly}
\phantomsection\label{the-first-stage-error-appears-explicitly}
\[
\left\|(I-\widehat\beta\widehat\beta^\top)\beta_0\right\|
=
O_p\!\left(
  h^3+h\delta_n+h^{-1}\delta_n^2+
  \focus{h^{-1}R_n}
\right).
\]
