% ---------- Footer ----------
% Hides numbering on the title and section-divider slides.
% Subtracts the title slide so the first content slide is 1/N.
\setbeamertemplate{footline}{%
  \leavevmode

  \ifnum\value{framenumber}>1
    \hbox{%
      \begin{beamercolorbox}[
        wd=\paperwidth,
        ht=0.28cm,
        dp=0.16cm,
        rightskip=0.55cm
      ]{page number in head/foot}
        \hfill
        \the\numexpr\value{framenumber}-1\relax
        \,/\,%
        \the\numexpr\inserttotalframenumber-1\relax
      \end{beamercolorbox}
    }
  \fi
}
